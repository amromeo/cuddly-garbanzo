function status(t){if(hasBlocker(t))return 'blocked';if(t.deferred)return 'deferred';if(approved(t))return 'approved';if(unresolved(t))return 'review';if(t.reviewFields.some(k=>t.decisions[k]==='keep'&&t.proposed[k]!==t.epic[k]))return 'retained';if(t.draft)return 'draft';return 'draft';}
const statusLabel={review:'Needs review',blocked:'Blocked',draft:'Draft saved',retained:'Wording retained',approved:'Approved',deferred:'Deferred'};
function workflowBadge(t){const code=t===working&&dirty?'unsaved':approved(t)?'approved':t.draft?'draft':'pending';const label=code==='unsaved'?'Unsaved changes':code==='approved'?'Approved':code==='draft'?'Draft saved':'Not yet saved';return `<span class="badge ${code}" data-workflow-status="${code}">${label}</span>`;}
function reviewBadge(t){let code,label;if(hasBlocker(t)){code='blocked';label='Blocked';}else if(t.deferred){code='deferred';label='Deferred';}else if(unresolved(t)&&!approved(t)){code='review';label='Needs review';}else if(retainedDifference(t)){code='retained';label='Wording retained';}else{code='complete';label='Review complete';}return `<span class="badge ${code}" data-review-status="${code}">${label}</span>`;}
function badge(t){return `<div class="status-pair"><span><small>Save / approval</small>${workflowBadge(t)}</span><span><small>Review</small>${reviewBadge(t)}</span></div>`;}
function changeSummary(t){if(hasBlocker(t))return t.reason;if(retainedDifference(t)&&!unresolved(t))return 'Catalog wording intentionally differs from Epic';if(t.deferred)return 'Source change deferred for later review';if(approved(t))return 'Approved content ready to export';return t.reason;}
function includedKeys(t){return FIELDS.map(f=>f.key).filter(k=>t.include.includes(k));}
